Introduction

This document downloads the AB-Bind database into data/_raw/. The dataset contains 32 antibody-antigen complex structures, the experimental binding measurements (AB-Bind_experimental_data.csv) and computational binding energy estimates (ABbind_compt_data.zip).

Download the data

# Download the repository as a zip from GitHub into a temp file
repo_zip_url <- str_c(
  "https://codeload.github.com/sarahsirin/AB-Bind-Database/",
  "zip/refs/heads/master"
)
tmp_zip <- tempfile(fileext = ".zip")
curl_download(url = repo_zip_url, destfile = tmp_zip)

# Unpack the zip in a temp dir
tmp_dir <- tempfile()
dir.create(tmp_dir)
unzip(tmp_zip, exdir = tmp_dir)
repo_dir <- setdiff(
  list.dirs(tmp_dir, recursive = FALSE),
  file.path(tmp_dir, "__MACOSX")
)

# Copy the repository files into data/_raw/ and unzip the computational data
raw_dir <- here("data/_raw")
dir.create(raw_dir, recursive = TRUE, showWarnings = FALSE)
file.copy(
  list.files(repo_dir, full.names = TRUE),
  raw_dir,
  overwrite = TRUE
)

Computational binding data

The unpacked zip holds one .dat file per method, semicolon-delimited with the columns pdb, mutation, expt and comp.

# Read all .dat files
compt_files <- list.files(
  here("data/_raw", "ABbind_compt_data"),
  pattern = "\\.dat$",
  full.names = TRUE
)

compt_data <- compt_files |>
  set_names(~ str_remove(basename(.x), "\\.dat$")) |>
  map_dfr(
    ~ read_delim(
      .x,
      delim = ";",
      comment = "#",
      trim_ws = TRUE,
      col_names = c("pdb", "mutation", "expt", "comp"),
      col_types = cols(
        pdb = col_character(),
        mutation = col_character(),
        expt = col_double(),
        comp = col_double()
      ),
      show_col_types = FALSE
    ),
    .id = "method"
  )

Save the loaded data

The experimental and computational binding data are saved as two CSV files in data/. Both files are read in 02_clean.qmd for the cleaning step.

  x = ab_bind,
  file = here("data", "01_dat_load_experimental.csv")
)
